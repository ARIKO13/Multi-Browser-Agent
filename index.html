Tampermonkey Auto-Inject Script (Auto-Paste & Auto-Send)

Karena keamanan browser melarang website mengisi kolom chat tab lain secara langsung, pasang script ringan ini di ekstensi Tampermonkey (Chrome/Edge/Brave). Setiap kali hub ini membuka tab Gemini, DeepSeek, atau GLM dengan parameter #nexus_prompt=..., script ini otomatis mengetikkan prompt-nya dan menekan tombol kirim untukmu! 

// ==UserScript==
// @name         Nexus Multi-Browser AI Auto-Relay Injector
// @namespace    http://tampermonkey.net/
// @version      1.0
// @description  Otomatis isi & kirim prompt estafet dari Nexus Multi-Agent Hub
// @match        https://gemini.google.com/*
// @match        https://chat.deepseek.com/*
// @match        https://chatglm.cn/*
// @match        https://z.ai/*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';
    const hash = window.location.hash;
    if (!hash.includes('nexus_prompt=')) return;
    const rawPrompt = decodeURIComponent(hash.split('nexus_prompt=')[1].split('&')[0]);
    if (!rawPrompt) return;

    const tryInject = setInterval(() => {
        const editor = document.querySelector('div[contenteditable="true"], textarea');
        if (editor) {
            clearInterval(tryInject);
            editor.focus();
            if (editor.tagName === 'TEXTAREA') {
                const nativeSetter = Object.getOwnPropertyDescriptor(window.HTMLTextAreaElement.prototype, 'value').set;
                nativeSetter.call(editor, rawPrompt);
                editor.dispatchEvent(new Event('input', { bubbles: true }));
            } else {
                editor.innerText = rawPrompt;
                editor.dispatchEvent(new Event('input', { bubbles: true }));
            }
            setTimeout(() => {
                const sendBtn = document.querySelector('button[aria-label*="Send"], button[aria-label*="Kirim"], button[type="submit"]');
                if (sendBtn && !sendBtn.disabled) sendBtn.click();
                history.replaceState(null, '', window.location.pathname + window.location.search);
            }, 900);
        }
    }, 800);
})();
